Python Web Scraper Script for ft.lk

Here is the ready-to-run Python script using BeautifulSoup & requests to fetch articles from Daily FT:

import requests
from bs4 import BeautifulSoup
import json
import time

BASE_URL = "https://www.ft.lk"
HEADERS = {
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36"
}

def scrape_daily_ft_business():
    print("[+] Scraping Daily FT Business & Governance articles...")
    url = f"{BASE_URL}/financial-services/42"
    response = requests.get(url, headers=HEADERS, timeout=10)
    
    if response.status_code != 200:
        print(f"[-] Failed to fetch page: Status {response.status_code}")
        return []

    soup = BeautifulSoup(response.content, 'html.parser')
    articles = []
    
    # Target Daily FT news card containers
    news_items = soup.find_all('div', class_='news-card') or soup.find_all('div', class_='col-md-6')
    
    for item in news_items:
        title_tag = item.find('h3') or item.find('h4') or item.find('a')
        link_tag = item.find('a', href=True)
        img_tag = item.find('img', src=True)
        desc_tag = item.find('p')
        
        if title_tag and link_tag:
            title = title_tag.get_text(strip=True)
            link = link_tag['href'] if link_tag['href'].startswith('http') else BASE_URL + link_tag['href']
            img = img_tag['src'] if img_tag else ""
            desc = desc_tag.get_text(strip=True) if desc_tag else ""
            
            articles.append({
                "title": title,
                "link": link,
                "image": img,
                "summary": desc,
                "source": "Daily FT (ft.lk)",
                "category": "Business & Governance",
                "scraped_at": time.strftime("%Y-%m-%d %H:%M:%S")
            })
            
    print(f"[+] Scraped {len(articles)} articles successfully.")
    return articles

if __name__ == "__main__":
    results = scrape_daily_ft_business()
    print(json.dumps(results[:2], indent=2))
